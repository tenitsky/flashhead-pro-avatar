# ------------------------------------------------------------------------------
# 4. STORAGE
# ------------------------------------------------------------------------------

Container disk:           20 GB    # Global Volume: first-boot downloads are staged here
                                   # (Network Volume alone would need only 5 GB)
Volume disk:              0 GB     # do NOT use a pod-local volume disk
Volume mount path:        /workspace

# At deploy time, each user attaches their OWN volume at /workspace, either kind:
#
#   Network Volume (regional)
#     Recommended size: 50 GB (models ~6.9 GB, plus ComfyUI, venv, renders).
#     Everything lives on it. To replace a pod: deploy in the same datacenter,
#     attach the same volume.
#
#   Global Volume (beta, any datacenter, grows by itself)
#     Models, inputs, outputs and saved workflows live on it. ComfyUI, its venv,
#     the node pack and upstream's code run from the container disk, set up again
#     on each fresh container (a few minutes). Let the first pod finish downloading
#     before starting more pods on the same volume.
#
# setup.sh detects which one is attached (log line "Storage: network volume ..." or
# "Storage: global volume ...") and refuses to run on the container disk alone.
# Override with FLASHHEAD_STORAGE=network or FLASHHEAD_STORAGE=global.
# Storage types: https://docs.runpod.io/pods/storage/types
# Global Volumes: https://docs.runpod.io/storage/globalvolume/overview

# ------------------------------------------------------------------------------
# 6. ENVIRONMENT VARIABLES
# ------------------------------------------------------------------------------
# Put these in the template:

JUPYTER_NO_AUTH=1
# 1 = Jupyter with no login (anyone with the URL gets a shell).
# For a login: JUPYTER_NO_AUTH=0 AND JUPYTER_PASSWORD=<token>.

HF_TOKEN=
# Optional. Both model repos are ungated; a token only raises HF rate limits.
# Better as a RunPod secret:  HF_TOKEN={{ RUNPOD_SECRET_hf_token }}

# Optional, normally left out of the template:
#   JUPYTER_PASSWORD=          Jupyter token when JUPYTER_NO_AUTH=0
#   FLASHHEAD_REF=             test a newer SoulX-FlashHead commit (engine built on the pin)
#   FILEBROWSER_PASSWORD=      only if port 8080 is exposed
#   HF_HOME=                   default /workspace/.cache/huggingface
#   FLASHHEAD_STORAGE=         network or global: override the volume detection
#   TEMPLATE_REF=              branch or tag to clone (default main;
#                              v1-network-volume = before Global Volume support)
#
# Never set COMFYUI_PATH: the image starts ComfyUI at /workspace/runpod-slim/ComfyUI.
